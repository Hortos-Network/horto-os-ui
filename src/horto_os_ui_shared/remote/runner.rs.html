22/// Options for a remote OpenSSH session and binary source.
23#[derive(Debug, Clone)]
24pub struct RemoteOptions {
25    /// OpenSSH Host alias or `user@host`.
26    pub host: String,
27    /// When true, run `ssh-copy-id` once before apply. Default false.
28    pub install_ssh_key: bool,
29    /// Local directory with the three box binaries (skips GitHub download).
30    pub bin_dir: Option<PathBuf>,
31    /// Workspace / Release version (`0.1.0`, without `v`).
32    pub version: String,
33    /// `owner/repo` for Release downloads.
34    pub github_repo: String,
35    /// Force `SSH_ASKPASS` (Desktop / no TTY).
36    pub force_askpass: bool,
37    /// Cache root for downloaded Release assets.
38    pub cache_root: PathBuf,
39    /// Remote temp dir for the apply agent.
40    pub remote_agent_dir: String,
41    /// Remote install dir for permanent binaries.
42    pub install_dir: String,
43    /// Optional OpenSSH config file (`ssh -F`). Used by tests and custom layouts.
44    pub ssh_config_file: Option<PathBuf>,
45}
46

64/// CLI arguments to run on the box via the uploaded agent (without the binary name).
65#[derive(Debug, Clone)]
66pub struct RemoteRunRequest {
67    /// Session and binary options.
68    pub options: RemoteOptions,
69    /// Arguments after the remote `horto-os-ui` binary (e.g. `["--dry-run", "setup", "run", "--full"]`).
70    pub cli_args: Vec<String>,
71    /// Prefix with `sudo -n` / `sudo` when true (apply mode).
72    pub use_sudo: bool,
73    /// After a successful command, install CLI+TUI+API and enable the API unit.
74    pub install_payload_on_success: bool,
75}

77fn session_from(opts: &RemoteOptions) -> Result<SshSession> {
78    Ok(SshSession {
79        host: parse_host_spec(&opts.host)?,
80        env: SshEnv {
81            force_askpass: opts.force_askpass,
82        },
83        config_file: opts.ssh_config_file.clone(),
84    })
85}

87fn shell_quote(arg: &str) -> String {
88    if arg.is_empty() {
89        return "''".to_owned();
90    }
91    if arg
92        .chars()
93        .all(|c| c.is_ascii_alphanumeric() || matches!(c, '-' | '_' | '.' | '/' | '=' | ':' | '+'))
94    {
95        return arg.to_owned();
96    }
97    format!("'{}'", arg.replace('\'', "'\\''"))
98}

100fn remote_agent_bin(opts: &RemoteOptions) -> String {
101    format!(
102        "{}/horto-os-ui",
103        opts.remote_agent_dir.trim_end_matches('/')
104    )
105}

107fn build_remote_command(opts: &RemoteOptions, cli_args: &[String], use_sudo: bool) -> String {
108    let bin = remote_agent_bin(opts);
109    let mut parts = Vec::new();
110    if use_sudo {
111        parts.push("sudo".to_owned());
112    }
113    parts.push(shell_quote(&bin));
114    for a in cli_args {
115        parts.push(shell_quote(a));
116    }
117    parts.join(" ")
118}

120/// Probe `uname -m` on the remote host.
121///
122/// # Errors
123///
124/// Returns [`crate::HortoError`] when SSH fails or the arch is unsupported.
125pub fn remote_probe_arch(runner: &dyn ProcessRunner, opts: &RemoteOptions) -> Result<BoxArch> {
126    let session = session_from(opts)?;
127    let out = session.exec(runner, "uname -m", StdioMode::Capture)?;
128    box_arch_from_uname(&out.stdout)
129}

131fn ensure_agent(
132    runner: &dyn ProcessRunner,
133    session: &SshSession,
134    opts: &RemoteOptions,
135    bins: &LocalBins,
136) -> Result<()> {
137    session.exec(
138        runner,
139        &format!("mkdir -p {}", shell_quote(&opts.remote_agent_dir)),
140        StdioMode::Capture,
141    )?;
142    let remote_bin = remote_agent_bin(opts);
143    session.scp_to(runner, &bins.cli, &remote_bin)?;
144    session.exec(
145        runner,
146        &format!("chmod +x {}", shell_quote(&remote_bin)),
147        StdioMode::Capture,
148    )?;
149    Ok(())
150}

152fn maybe_install_key(
153    runner: &dyn ProcessRunner,
154    session: &SshSession,
155    opts: &RemoteOptions,
156) -> Result<()> {
157    if opts.install_ssh_key {
158        session.install_ssh_key(runner)?;
159    }
160    Ok(())
161}

163/// Upload the CLI agent (if needed), optionally install a public key, run a remote CLI command.
164///
165/// # Errors
166///
167/// Returns [`crate::HortoError`] on SSH/SCP/agent failures.
168pub fn remote_run_cli(runner: &dyn ProcessRunner, req: &RemoteRunRequest) -> Result<String> {
169    let opts = &req.options;
170    let session = session_from(opts)?;
171    let arch = {
172        let out = session.exec(runner, "uname -m", StdioMode::Capture)?;
173        box_arch_from_uname(&out.stdout)?
174    };
175    let bins = ensure_local_bins(
176        runner,
177        &opts.version,
178        &opts.github_repo,
179        arch,
180        opts.bin_dir.as_deref(),
181        &opts.cache_root,
182    )?;
183    ensure_agent(runner, &session, opts, &bins)?;
184    maybe_install_key(runner, &session, opts)?;
185
186    let remote_cmd = build_remote_command(opts, &req.cli_args, req.use_sudo);
187    // Inherit stdio so SSH/sudo password prompts work on a TTY (CLI/TUI).
188    let out = session.exec(runner, &remote_cmd, StdioMode::Inherit)?;
189    let mut log = out.stdout.clone();
190    if !out.stderr.trim().is_empty() {
191        if !log.is_empty() {
192            log.push('\n');
193        }
194        log.push_str(&out.stderr);
195    }
196    if log.trim().is_empty() {
197        log = format!("remote command finished: {remote_cmd}");
198    }
199
200    if req.install_payload_on_success {
201        remote_install_payload(runner, opts, &bins)?;
202    }
203    Ok(log)
204}

206/// Convenience: remote `setup run` with optional payload install.
207///
208/// # Errors
209///
210/// Returns [`crate::HortoError`] when the remote setup fails.
211pub fn remote_setup_run(
212    runner: &dyn ProcessRunner,
213    opts: RemoteOptions,
214    dry_run: bool,
215    full: bool,
216    skip_piper: bool,
217    install_payload: bool,
218) -> Result<String> {
219    let mut cli_args = Vec::new();
220    if dry_run {
221        cli_args.push("--dry-run".into());
222    }
223    if skip_piper {
224        cli_args.push("--skip-piper".into());
225    }
226    cli_args.push("setup".into());
227    cli_args.push("run".into());
228    if full {
229        cli_args.push("--full".into());
230    } else {
231        cli_args.push("--minimal".into());
232    }
233    remote_run_cli(
234        runner,
235        &RemoteRunRequest {
236            options: opts,
237            cli_args,
238            use_sudo: !dry_run,
239            install_payload_on_success: install_payload && !dry_run,
240        },
241    )
242}

257/// Copy CLI + TUI + API into `install_dir` and enable the status-api systemd unit.
258///
259/// # Errors
260///
261/// Returns [`crate::HortoError`] when transfer or remote enable fails.
262pub fn remote_install_payload(
263    runner: &dyn ProcessRunner,
264    opts: &RemoteOptions,
265    bins: &LocalBins,
266) -> Result<()> {
267    let session = session_from(opts)?;
268    let staging = format!("{}/payload", opts.remote_agent_dir.trim_end_matches('/'));
269    session.exec(
270        runner,
271        &format!("mkdir -p {}", shell_quote(&staging)),
272        StdioMode::Capture,
273    )?;
274
275    let locals: [&Path; 3] = [&bins.cli, &bins.tui, &bins.status_api];
276    transfer_files(runner, &session, &locals, &staging)?;
277
278    let install = opts.install_dir.trim_end_matches('/');
279    let unit_path = "/etc/systemd/system/horto-os-ui-status-api.service";
280    let write_unit = format!(
281        "sudo tee {unit_path} > /dev/null <<'HORTO_UNIT_EOF'\n{STATUS_API_UNIT}HORTO_UNIT_EOF"
282    );
283    let move_bins = format!(
284        "sudo install -m 755 {staging}/horto-os-ui {staging}/horto-os-ui-tui {staging}/horto-os-ui-status-api {install}/ && \
285         sudo systemctl daemon-reload && \
286         sudo systemctl enable --now horto-os-ui-status-api.service"
287    );
288    session.exec(runner, &write_unit, StdioMode::Inherit)?;
289    session.exec(runner, &move_bins, StdioMode::Inherit)?;
290
291    if install != "/usr/local/bin" {
292        let fix = format!(
293            "sudo sed -i 's|/usr/local/bin/horto-os-ui-status-api|{install}/horto-os-ui-status-api|' {unit_path} && \
294             sudo systemctl daemon-reload && sudo systemctl restart horto-os-ui-status-api.service"
295        );
296        session.exec(runner, &fix, StdioMode::Inherit)?;
297    }
298    Ok(())
299}
