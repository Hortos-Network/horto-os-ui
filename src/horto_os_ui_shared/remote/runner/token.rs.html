1//! Local status-api bearer token save path and prompts.
2
3use super::reboot::wants_reboot_now;
4use crate::error::{HortoError, Result};
5use std::path::PathBuf;
6
7/// Minimum tip bearer length. Box install writes 64 hex chars; shorter leftovers
8/// (including unit-test junk) must never be treated as a real Status API token.
9pub const MIN_API_TOKEN_HEX_LEN: usize = 32;
10
11/// Return trimmed hex when it is usable as a tip Status API bearer.
12///
13/// Accepts `HORTO_API_TOKEN=<hex>` or bare hex. Rejects empty, non-hex, and
14/// anything shorter than [`MIN_API_TOKEN_HEX_LEN`].
15#[must_use]
16pub fn usable_api_token_hex(raw: &str) -> Option<&str> {
17    let hex = raw.strip_prefix("HORTO_API_TOKEN=").unwrap_or(raw).trim();
18    if hex.len() < MIN_API_TOKEN_HEX_LEN || !hex.chars().all(|c| c.is_ascii_hexdigit()) {
19        None
20    } else {
21        Some(hex)
22    }
23}
24
25/// Parse drop-file contents: `HORTO_API_TOKEN=<hex>` or bare hex.
26#[must_use]
27pub fn parse_api_token_drop(raw: &str) -> Option<String> {
28    let line = raw.lines().map(str::trim).find(|l| !l.is_empty())?;
29    usable_api_token_hex(line).map(str::to_owned)
30}

32pub fn api_token_config_path() -> PathBuf {
33    if let Ok(xdg) = std::env::var("XDG_CONFIG_HOME") {
34        let trimmed = xdg.trim();
35        if !trimmed.is_empty() {
36            return PathBuf::from(trimmed).join("horto-os-ui").join("api_token");
37        }
38    }
39    let home = std::env::var("HOME").unwrap_or_else(|_| ".".into());
40    PathBuf::from(home)
41        .join(".config")
42        .join("horto-os-ui")
43        .join("api_token")
44}

46/// Write hex bearer to the local config path with mode `0o600`.
47///
48/// Rejects short / non-hex input **before** creating or truncating the tip file,
49/// so a bad caller cannot wipe a real PC token with junk.
50///
51/// # Errors
52///
53/// Returns [`crate::HortoError`] when the token is unusable, or the directory /
54/// file cannot be written.
55pub fn write_api_token_file(token: &str) -> Result<PathBuf> {
56    use std::fs;
57    use std::io::Write;
58
59    let hex = usable_api_token_hex(token).ok_or_else(|| {
60        HortoError::msg(format!(
61            "status-api bearer must be at least {MIN_API_TOKEN_HEX_LEN} hex characters"
62        ))
63    })?;
64
65    let path = api_token_config_path();
66    if let Some(parent) = path.parent() {
67        fs::create_dir_all(parent)
68            .map_err(|e| HortoError::msg(format!("create {}: {e}", parent.display())))?;
69    }
70    {
71        #[cfg(unix)]
72        {
73            use std::os::unix::fs::OpenOptionsExt;
74            let mut f = fs::OpenOptions::new()
75                .write(true)
76                .create(true)
77                .truncate(true)
78                .mode(0o600)
79                .open(&path)
80                .map_err(|e| HortoError::msg(format!("write {}: {e}", path.display())))?;
81            writeln!(f, "{hex}")
82                .map_err(|e| HortoError::msg(format!("write {}: {e}", path.display())))?;
83        }
84        #[cfg(not(unix))]
85        {
86            fs::write(&path, format!("{hex}\n"))
87                .map_err(|e| HortoError::msg(format!("write {}: {e}", path.display())))?;
88        }
89    }
90    Ok(path)
91}

93/// Finish a save-token prompt given the raw answer (test seam).
94///
95/// # Errors
96///
97/// Returns [`crate::HortoError`] when the file cannot be written.
98pub fn finish_save_api_token(token: &str, answer: &str) -> Result<bool> {
99    if !wants_reboot_now(answer) {
100        return Ok(false);
101    }
102    let path = write_api_token_file(token)?;
103    tracing::info!(
104        path = %path.display(),
105        "saved status-api bearer (hex only; not logged)"
106    );
107    Ok(true)
108}

110/// Propose saving the status-api bearer to `~/.config/horto-os-ui/api_token`.
111///
112/// TTY: `[y/N]` prompt. Non-TTY: skip write and log the hex once for paste.
113///
114/// # Errors
115///
116/// Returns [`crate::HortoError`] when stdin cannot be read or the file write fails.
117pub fn offer_save_api_token(token: &str) -> Result<bool> {
118    use std::io::IsTerminal;
119    offer_save_api_token_with(token, std::io::stdin().is_terminal(), None)
120}
121
122/// Testable core of [`offer_save_api_token`] (`canned_answer` skips stdin when `Some`).
123///
124/// # Errors
125///
126/// Returns [`crate::HortoError`] when stdin cannot be read or the file write fails.
127pub fn offer_save_api_token_with(
128    token: &str,
129    is_tty: bool,
130    canned_answer: Option<&str>,
131) -> Result<bool> {
132    use std::io::{self, Write};
133
134    if !is_tty {
135        tracing::info!(
136            token,
137            "status-api bearer (non-TTY; paste into Desktop Connection or save manually)"
138        );
139        return Ok(false);
140    }
141    eprint!("Save status-api bearer to ~/.config/horto-os-ui/api_token? [y/N]: ");
142    let _ = io::stderr().flush();
143    let line = if let Some(answer) = canned_answer {
144        answer.to_owned()
145    } else {
146        let mut line = String::new();
147        io::stdin()
148            .read_line(&mut line)
149            .map_err(|e| HortoError::msg(format!("read token save prompt: {e}")))?;
150        line
151    };
152    finish_save_api_token(token, &line)
153}
