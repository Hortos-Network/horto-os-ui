9/// Issue `sudo reboot` on the box (CLI: password on the terminal via Inherit).
10///
11/// # Errors
12///
13/// Returns [`crate::HortoError`] when SSH fails before reboot starts.
14pub fn remote_reboot(runner: &dyn ProcessRunner, opts: &RemoteOptions) -> Result<()> {
15    let session = session_from(opts)?;
16    finish_remote_reboot(runner, &session, "y", None)
17}

19/// Issue `sudo reboot` using a sudo password already collected by the UI.
20///
21/// Feeds `sudo -S` over captured SSH. Used by the TUI (no cooked TTY).
22///
23/// # Errors
24///
25/// Returns [`crate::HortoError`] when SSH fails before reboot starts.
26pub fn remote_reboot_with_sudo_password(
27    runner: &dyn ProcessRunner,
28    opts: &RemoteOptions,
29    sudo_password: &str,
30) -> Result<()> {
31    let session = session_from(opts)?;
32    finish_remote_reboot(runner, &session, "y", Some(sudo_password))
33}

35/// Whether a reboot prompt answer means reboot now.
36#[must_use]
37pub fn wants_reboot_now(raw: &str) -> bool {
38    matches!(raw.trim().to_ascii_lowercase().as_str(), "y" | "yes")
39}

41/// Prompt on a local TTY, then `sudo reboot` on the box. Non-TTY prints a reminder.
42pub fn offer_remote_reboot(runner: &dyn ProcessRunner, session: &SshSession) -> Result<()> {
43    use std::io::{self, IsTerminal, Write};
44
45    if !io::stdin().is_terminal() {
46        eprintln!("Reboot recommended for hostname/network changes. On the box: sudo reboot");
47        return Ok(());
48    }
49    eprint!("Reboot the box now to apply hostname/network changes? [y/N]: ");
50    let _ = io::stderr().flush();
51    let mut line = String::new();
52    io::stdin()
53        .read_line(&mut line)
54        .map_err(|e| crate::error::HortoError::msg(format!("read reboot prompt: {e}")))?;
55    finish_remote_reboot(runner, session, &line, None)
56}

58pub fn finish_remote_reboot(
59    runner: &dyn ProcessRunner,
60    session: &SshSession,
61    answer: &str,
62    sudo_password: Option<&str>,
63) -> Result<()> {
64    if !wants_reboot_now(answer) {
65        eprintln!("Skipping reboot. Reboot the box later when convenient.");
66        return Ok(());
67    }
68    remote_progress(
69        &session.host.raw,
70        "reboot box (SSH + sudo; may ask password)",
71    );
72    eprintln!("Rebooting...");
73    let used_stdin = sudo_password.is_some();
74    let result = sudo_password.map_or_else(
75        || session.exec(runner, "sudo reboot", StdioMode::Inherit),
76        |pass| {
77            let mut feed = String::with_capacity(pass.len() + 1);
78            feed.push_str(pass);
79            feed.push('\n');
80            let out = session.exec_stdin_reboot(runner, "sudo -S -p '' reboot", feed.as_bytes());
81            wipe_secret(&mut feed);
82            out
83        },
84    );
85    match result {
86        Ok(_) => Ok(()),
87        Err(e) => {
88            if used_stdin && !ssh_drop_after_reboot(&e) {
89                return Err(e);
90            }
91            // Host drop mid-session is expected once reboot starts.
92            eprintln!("reboot issued (SSH session closed is expected): {e}");
93            Ok(())
94        }
95    }
96}

98pub fn ssh_drop_after_reboot(err: &crate::error::HortoError) -> bool {
99    let s = err.to_string().to_ascii_lowercase();
100    s.contains("connection closed")
101        || s.contains("connection reset")
102        || s.contains("broken pipe")
103        || s.contains("exit 255")
104}
