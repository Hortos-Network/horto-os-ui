24pub fn upload_remote_cli(
25    runner: &dyn ProcessRunner,
26    session: &SshSession,
27    opts: &RemoteOptions,
28    bins: &LocalBins,
29) -> Result<String> {
30    session.exec(
31        runner,
32        &format!("mkdir -p {}", shell_quote(&opts.remote_agent_dir)),
33        StdioMode::Capture,
34    )?;
35    let remote_bin = remote_agent_bin(opts);
36    remote_progress(
37        &opts.host,
38        "upload horto CLI to the box (SCP; may ask password)",
39    );
40    session.scp_to(runner, &bins.cli, &remote_bin)?;
41    session.exec(
42        runner,
43        &format!("chmod +x {}", shell_quote(&remote_bin)),
44        StdioMode::Capture,
45    )?;
46    Ok(remote_bin)
47}

59/// Upload the PC CLI to the box agent dir (s0 / explicit sync).
60///
61/// # Errors
62///
63/// Returns [`crate::HortoError`] on SSH/SCP/local-bin failures.
64pub fn remote_upload_cli(
65    runner: &dyn ProcessRunner,
66    opts: &RemoteOptions,
67) -> Result<RemoteCliProbe> {
68    let session = session_from(opts)?;
69    remote_progress(&opts.host, "probe arch (SSH; may ask password)");
70    let out = session.exec(runner, "uname -m", StdioMode::Capture)?;
71    let arch = box_arch_from_uname(&out.stdout)?;
72    let bins = ensure_local_bins(
73        runner,
74        &opts.release_tag,
75        &opts.version,
76        &opts.github_repo,
77        arch,
78        opts.bin_dir.as_deref(),
79        &opts.cache_root,
80    )?;
81    upload_remote_cli(runner, &session, opts, &bins)?;
82    maybe_install_key(runner, &session, opts)?;
83    probe_remote_cli(runner, &session, opts)
84}

86/// Ensure a current CLI on the box: reuse when version matches, otherwise SCP once.
87pub fn ensure_remote_cli(
88    runner: &dyn ProcessRunner,
89    session: &SshSession,
90    opts: &RemoteOptions,
91    bins: &LocalBins,
92) -> Result<String> {
93    let probe = probe_remote_cli(runner, session, opts)?;
94    if probe.current {
95        if let Some(path) = probe.path {
96            remote_progress(&opts.host, "CLI on box already current; skip upload");
97            return Ok(path);
98        }
99    }
100    upload_remote_cli(runner, session, opts, bins)
101}

116    runner: &dyn ProcessRunner,
117    opts: &RemoteOptions,
118) -> Result<(SshSession, LocalBins, String)> {

120    remote_progress(&opts.host, "probe arch (SSH; may ask password)");
121    let out = session.exec(runner, "uname -m", StdioMode::Capture)?;
122    let arch = box_arch_from_uname(&out.stdout)?;
123    let bins = ensure_local_bins(

128        arch,

130        &opts.cache_root,

146/// Drop tip-only globals so an older box agent can still parse argv.
147#[must_use]
148pub fn strip_tip_only_cli_flags(args: &[String]) -> Vec<String> {
149    args.iter()
150        .filter(|a| {
151            let s = a.as_str();
152            !(s == "--install-status-api"
153                || s == "--no-install-status-api"
154                || s == "--install-mcp"
155                || s == "--no-install-mcp"
156                || s == "--stacks"
157                || s.starts_with("--stacks="))
158        })
159        .cloned()
160        .collect()
161}

163fn stale_cli_error(box_label: &str) -> HortoError {
164    HortoError::msg(format!(
165        "Box CLI is out of date (box={box_label}, tip={LONG_VERSION}). Sync CLI first, refresh tip release or set HORTO_BIN_DIR to matching bins, or allow an old CLI."
166    ))

169/// Upload the CLI agent (if needed), optionally install a public key, run a remote CLI command.
170///
171/// Refuses when the box CLI long-version does not match tip [`LONG_VERSION`], unless
172/// [`RemoteRunFlags::allow_stale_cli`] is set. With allow + stale, tip-only argv flags
173/// (`--install-*` / `--stacks`) are stripped so older agents can still parse.
174///
175/// # Errors
176///
177/// Returns [`crate::HortoError`] on SSH/SCP/agent failures or a stale CLI without allow.
178pub fn remote_run_cli(
179    runner: &dyn ProcessRunner,
180    req: &RemoteRunRequest,
181) -> Result<RemoteRunOutcome> {
182    let opts = &req.options;
183    let (session, bins, remote_bin) = prepare_remote_agent(runner, opts)?;
184    let probe = probe_remote_cli(runner, &session, opts)?;
185    if !probe.current && !req.allow_stale_cli {
186        let label = probe
187            .version
188            .as_deref()
189            .unwrap_or_else(|| probe.status.as_label());
190        return Err(stale_cli_error(label));
191    }
192
193    let cli_args = if probe.current {
194        req.cli_args.clone()
195    } else {
196        strip_tip_only_cli_flags(&req.cli_args)
197    };
198    let sudo_kind = remote_sudo_kind(req.use_sudo, req.sudo_password.as_deref());
199    let remote_cmd = build_remote_command_sudo(&remote_bin, &cli_args, sudo_kind);
200    remote_progress(
201        &opts.host,
202        &remote_run_banner_detail(&remote_cmd, req.use_sudo),
203    );
204    // Desktop (sudo_password Some) never inherits the launch TTY.
205    let capture = req.capture_output || req.sudo_password.is_some();
206    let out = match sudo_kind {
207        RemoteSudoKind::Stdin => {
208            let pass = req.sudo_password.as_deref().unwrap_or("");
209            let mut feed = String::with_capacity(pass.len() + 1);
210            feed.push_str(pass);
211            feed.push('\n');
212            let result = session.exec_stdin(runner, &remote_cmd, feed.as_bytes());
213            wipe_secret(&mut feed);
214            result?
215        }
216        RemoteSudoKind::None | RemoteSudoKind::Prompt | RemoteSudoKind::NonInteractive => {
217            let stdio = if capture {
218                StdioMode::Capture
219            } else {
220                // Inherit so SSH/sudo password prompts work on a TTY (CLI/TUI apply).
221                StdioMode::Inherit
222            };
223            session.exec(runner, &remote_cmd, stdio)?
224        }
225    };
226    let secret = req.sudo_password.as_deref().unwrap_or("");
227    let log = redact_secret(&merge_command_log(&out, &remote_cmd), secret);
228    if capture && !log.trim().is_empty() {
229        publish_captured_apply_log(&log);
230    }
231
232    let install_payload = req.install_payload_on_success && req.ecosystem.any();
233    let api_token = if install_payload {
234        // Always push tip bins from the PC after apply. Do not gate on probe.current:
235        // Desktop tip SHA often differs from published Release bins; skipping left the
236        // box on stale /usr/local/bin while the agent already skipped ecosystem install.
237        remote_install_payload(
238            runner,
239            opts,
240            &bins,
241            req.ecosystem,
242            req.sudo_password.as_deref(),
243        )?
244    } else {
245        None
246    };
247    if req.offer_reboot_on_success {
248        offer_remote_reboot(runner, &session)?;
249    }
250    remote_progress(&opts.host, "remote setup finished");
251    Ok(RemoteRunOutcome { log, api_token })
252}

254/// Print Capture stdout once and mirror into the Desktop [`crate::log_bus::LogBus`] (when installed).
255pub fn publish_captured_apply_log(log: &str) {

259    }
260    eprintln!("{log}");

264/// Inputs for [`remote_setup_run`] (avoids a long bool parameter list).
265#[allow(clippy::struct_excessive_bools)]
266#[derive(Debug, Clone)]
267pub struct RemoteSetupRunArgs {
268    /// Session and binary options.
269    pub options: RemoteOptions,
270    /// Privileged apply on the box.
271    pub apply: bool,
272    /// Full pipeline when true; minimal when false.
273    pub full: bool,
274    /// Skip piper model download.
275    pub skip_piper: bool,
276    /// Ecosystem services to install after successful apply.
277    pub ecosystem: EcosystemInstallChoice,
278    /// Optional Docker stacks for d3.
279    pub stack_opts: crate::stack_opts::StackOpts,
280    /// Allow setup when box CLI ≠ tip long-version.
281    pub allow_stale_cli: bool,

294/// Convenience: remote `setup run` with optional payload install.
295///
296/// # Errors
297///
298/// Returns [`crate::HortoError`] when the remote setup fails or the box CLI is stale
299/// without [`RemoteSetupRunArgs::allow_stale_cli`].
300pub fn remote_setup_run(
301    runner: &dyn ProcessRunner,
302    args: RemoteSetupRunArgs,
303) -> Result<RemoteRunOutcome> {
304    let RemoteSetupRunArgs {
305        options: opts,
306        apply,
307        full,
308        skip_piper,
309        ecosystem,
310        stack_opts,
311        allow_stale_cli,
312        capture_output,
313        offer_reboot,
314        sudo_password,
315    } = args;
316    let mut cli_args = Vec::new();
317    if apply {
318        cli_args.push("--apply".into());
319    }
320    if skip_piper {
321        cli_args.push("--skip-piper".into());
322    }
323    let stacks_csv = stack_opts.to_csv();
324    if !stacks_csv.is_empty() {
325        cli_args.push(format!("--stacks={stacks_csv}"));
326    }
327    if ecosystem.status_api {
328        cli_args.push("--install-status-api".into());
329    } else {
330        cli_args.push("--no-install-status-api".into());
331    }
332    if ecosystem.mcp {
333        cli_args.push("--install-mcp".into());
334    } else {
335        cli_args.push("--no-install-mcp".into());
336    }
337    cli_args.push("setup".into());
338    cli_args.push("run".into());
339    if full {
340        cli_args.push("--full".into());
341    } else {
342        cli_args.push("--minimal".into());
343    }
344    remote_run_cli(
345        runner,
346        &RemoteRunRequest {
347            options: opts,
348            cli_args,
349            ecosystem,
350            flags: RemoteRunFlags {
351                use_sudo: apply,
352                install_payload_on_success: apply && ecosystem.any(),
353                offer_reboot_on_success: offer_reboot,
354                capture_output,
355                allow_stale_cli,
356            },
357            sudo_password,
358        },
359    )
360}

362/// Copy CLI + TUI + status-api (+ MCP when present) into `install_dir` and enable units.
363///
364/// Missing `horto-os-ui-mcp` skips MCP transfer/enable; other units still install.
365/// Returns the captured hex bearer when the drop file can be read.
366///
367/// # Errors
368///
369/// Returns [`crate::HortoError`] when transfer or remote enable fails.
370pub fn remote_install_payload(
371    runner: &dyn ProcessRunner,
372    opts: &RemoteOptions,
373    bins: &LocalBins,
374    choice: EcosystemInstallChoice,
375    sudo_password: Option<&str>,
376) -> Result<Option<String>> {
377    let choice = EcosystemInstallChoice {
378        status_api: choice.status_api,
379        mcp: choice.mcp && bins.mcp.is_some(),
380    };
381    if !choice.any() {
382        return Ok(None);
383    }
384    let session = session_from(opts)?;
385    let staging = format!("{}/payload", opts.remote_agent_dir.trim_end_matches('/'));
386    remote_progress(
387        &opts.host,
388        "copy apps onto the box (SCP/SSH; may ask password)",
389    );
390    session.exec(
391        runner,
392        &format!("mkdir -p {}", shell_quote(&staging)),
393        StdioMode::Capture,
394    )?;
395
396    let mut locals: Vec<&Path> = vec![&bins.cli, &bins.tui, &bins.status_api];
397    if let Some(ref mcp) = bins.mcp {
398        locals.push(mcp.as_path());
399    }
400    transfer_files(runner, &session, &locals, &staging)?;
401
402    let install = opts.install_dir.trim_end_matches('/');
403    remote_progress(
404        &opts.host,
405        "enable status-api and MCP on the box (SSH + sudo; may ask password)",
406    );
407    let enable = remote_enable_ecosystem_script(&staging, install, choice);
408    exec_payload_enable(runner, &session, &enable, sudo_password)?;
409
410    let mut api_token = None;
411    if choice.status_api {
412        let drop_path = format!("$HOME/{API_TOKEN_DROP_BASENAME}");
413        let cat_out = session.exec(
414            runner,
415            &format!("cat {drop_path} 2>/dev/null || true"),
416            StdioMode::Capture,
417        )?;
418        api_token = parse_api_token_drop(&cat_out.stdout);
419        let _ = session.exec(runner, &format!("rm -f {drop_path}"), StdioMode::Capture);
420        verify_remote_status_api(runner, &session, install)?;
421    }
422
423    Ok(api_token)
424}

426/// Run the multi-sudo enable script without a TTY.
427///
428/// With a password: `sudo -S bash -s` so every nested `sudo` in the script inherits
429/// the authenticated session (plain `sudo` after `sudo -S -v` still needs a TTY).
430fn exec_payload_enable(
431    runner: &dyn ProcessRunner,
432    session: &SshSession,
433    enable: &str,
434    sudo_password: Option<&str>,
435) -> Result<()> {
436    match sudo_password {
437        Some("") => {
438            session.exec_stdin(runner, "sudo -n bash -s", enable.as_bytes())?;
439        }
440        Some(pass) => {
441            let mut feed =
442                String::with_capacity(pass.len().saturating_add(1).saturating_add(enable.len()));
443            feed.push_str(pass);
444            feed.push('\n');
445            feed.push_str(enable);
446            let result = session.exec_stdin(runner, "sudo -S -p '' bash -s", feed.as_bytes());
447            wipe_secret(&mut feed);
448            result?;
449        }
450        None => {
451            session.exec(runner, enable, StdioMode::Inherit)?;
452        }
453    }
454    Ok(())
455}

457/// Confirm on-disk status-api `--version` matches the running `/health` `cli_version`.
458///
459/// # Errors
460///
461/// Returns [`crate::HortoError`] when `--version` fails, health fails, or the two disagree.
462fn verify_remote_status_api(
463    runner: &dyn ProcessRunner,
464    session: &SshSession,
465    install_dir: &str,
466) -> Result<()> {
467    remote_progress(
468        &session.host.raw,
469        "verify status-api on box (--version vs /health)",
470    );
471    let bin = format!("{install_dir}/horto-os-ui-status-api");
472    let ver_out = session.exec(
473        runner,
474        &format!("{} --version", shell_quote(&bin)),
475        StdioMode::Capture,
476    )?;
477    let health_out = session.exec(
478        runner,
479        "curl -fsS --connect-timeout 3 --max-time 8 http://127.0.0.1:8787/health",
480        StdioMode::Capture,
481    )?;
482    evaluate_status_api_verify(bin.as_str(), &ver_out, &health_out).map_err(|e| {
483        tracing::error!("status-api verify failed: {e}");
484        e
485    })?;
486    remote_progress(
487        &session.host.raw,
488        "status-api verify ok (--version matches /health)",
489    );
490    Ok(())
491}

493/// Check captured `--version` and `/health` outputs agree on the installed build.
494fn evaluate_status_api_verify(
495    bin: &str,
496    ver_out: &CommandOutput,
497    health_out: &CommandOutput,
498) -> Result<()> {
499    let on_disk = normalize_cli_version(&ver_out.stdout);
500    if !ver_out.success() || on_disk.is_empty() {
501        return Err(HortoError::msg(format!(
502            "status-api install verify failed: `{bin} --version` empty or non-zero (exit {})",
503            ver_out.status
504        )));
505    }
506    if !health_out.success() {
507        let detail = if health_out.stderr.trim().is_empty() {
508            health_out.stdout.trim()
509        } else {
510            health_out.stderr.trim()
511        };
512        return Err(HortoError::msg(format!(
513            "status-api install verify failed: GET http://127.0.0.1:8787/health (exit {}): {detail}",
514            health_out.status
515        )));
516    }
517    let Some(running) = health_cli_version(&health_out.stdout) else {
518        return Err(HortoError::msg(format!(
519            "status-api install verify failed: /health missing cli_version (got {})",
520            health_out.stdout.trim()
521        )));
522    };
523    if !status_api_versions_agree(&on_disk, &running) {
524        return Err(HortoError::msg(format!(
525            "status-api install verify failed: on-disk `{on_disk}` != running /health cli_version `{running}`"
526        )));
527    }
528    Ok(())
529}

531/// True when on-disk `--version` and `/health` `cli_version` name the same build.
532///
533/// Clap prints `horto-os-ui-status-api 0.1.0 (abc)`; health uses `0.1.0 (abc)`.
534fn status_api_versions_agree(on_disk: &str, health_cli: &str) -> bool {
535    let disk = normalize_cli_version(on_disk);
536    let health = normalize_cli_version(health_cli);
537    if disk.is_empty() || health.is_empty() {
538        return false;
539    }
540    disk == health || disk.ends_with(&format!(" {health}")) || disk.ends_with(&health)
541}

543/// Parse non-empty `cli_version` from a status-api `/health` JSON body.
544fn health_cli_version(body: &str) -> Option<String> {
545    let start = body.find('{')?;
546    let value: serde_json::Value = serde_json::from_str(body[start..].trim()).ok()?;
547    value
548        .get("cli_version")
549        .and_then(|v| v.as_str())
550        .map(str::trim)
551        .filter(|s| !s.is_empty())
552        .map(ToOwned::to_owned)
553}
