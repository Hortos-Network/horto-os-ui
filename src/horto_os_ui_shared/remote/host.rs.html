2//!
3//! Lists LAN names from `/etc/hosts`, plus OpenSSH `Host` aliases whose
4//! `HostName` is private or already named in `/etc/hosts` (no internet aliases).

13/// Destination passed to `ssh` / `scp` as the remote operand.
14#[derive(Debug, Clone, PartialEq, Eq)]
15pub struct HostSpec {
16    /// Raw value suitable for `ssh <spec>` (Host alias or `user@host`).
17    pub raw: String,
18}
19
20/// A hostname known from `/etc/hosts` and/or OpenSSH config (plus localhost).
21#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
22#[serde(rename_all = "camelCase")]
23pub struct KnownRemoteHost {
24    /// Name suitable as an OpenSSH target (`ssh <name>`) or Status API host.
25    pub name: String,
26    /// Present as a non-loopback entry in `/etc/hosts` (or synthetic localhost).
27    pub in_hosts_file: bool,
28    /// Present as a concrete `Host` alias in OpenSSH config (LAN-qualified).
29    pub in_ssh_config: bool,
30}
31
32/// One OpenSSH `Host` alias with its effective `HostName` (defaults to the alias).
33#[derive(Debug, Clone, PartialEq, Eq)]
34pub struct SshConfigHost {
35    /// `Host` token (no wildcards).
36    pub alias: String,
37    /// `HostName` value, or the alias when omitted.
38    pub host_name: String,
39}

41/// Parse a non-empty Host alias or `user@host` string.
42///
43/// # Errors
44///
45/// Returns [`crate::HortoError::Message`] when the string is empty or only whitespace.
46pub fn parse_host_spec(input: &str) -> Result<HostSpec> {
47    let raw = input.trim().to_owned();
48    if raw.is_empty() {
49        return Err(HortoError::msg(
50            "remote host is empty; pass an OpenSSH Host alias or user@host",
51        ));
52    }
53    if raw.contains(char::is_whitespace) {
54        return Err(HortoError::msg(format!(
55            "remote host must not contain whitespace: {raw:?}"
56        )));
57    }
58    Ok(HostSpec { raw })
59}

61/// Merge `/etc/hosts` LAN names with LAN-qualified OpenSSH `Host` aliases.
62///
63/// Always includes `localhost` first (this machine / Desktop local Status API).
64/// Missing files yield empty LAN sides (not an error). Names are sorted after
65/// localhost; duplicates keep both source flags.
66///
67/// # Errors
68///
69/// Returns [`crate::HortoError`] when a readable file cannot be read as UTF-8 text.
70pub fn list_known_remote_hosts() -> Result<Vec<KnownRemoteHost>> {
71    list_known_remote_hosts_from(Path::new("/etc/hosts"), &ssh_config_paths())
72}

74/// Testable merge of hosts-file text and SSH config texts (LAN filter applied).
75///
76/// `localhost` is always first so Desktop can target this machine.
77#[must_use]
78pub fn merge_known_remote_hosts(
79    hosts_file_text: &str,
80    ssh_config_texts: &[&str],
81) -> Vec<KnownRemoteHost> {
82    let hosts_names = parse_hosts_file_names(hosts_file_text);
83    let mut by_name: BTreeMap<String, KnownRemoteHost> = BTreeMap::new();
84    for name in &hosts_names {
85        by_name.insert(
86            name.clone(),
87            KnownRemoteHost {
88                name: name.clone(),
89                in_hosts_file: true,
90                in_ssh_config: false,
91            },
92        );
93    }
94    for text in ssh_config_texts {
95        for entry in parse_ssh_config_hosts(text) {
96            if !ssh_host_is_lan(&entry, &hosts_names) {
97                continue;
98            }
99            by_name
100                .entry(entry.alias.clone())
101                .and_modify(|h| h.in_ssh_config = true)
102                .or_insert(KnownRemoteHost {
103                    name: entry.alias,
104                    in_hosts_file: false,
105                    in_ssh_config: true,
106                });
107        }
108    }
109    with_localhost_first(by_name.into_values().collect())
110}
111
112/// Put `localhost` first for local Desktop targeting; drop duplicates.
113fn with_localhost_first(mut hosts: Vec<KnownRemoteHost>) -> Vec<KnownRemoteHost> {
114    hosts.retain(|h| h.name != "localhost");
115    let mut out = Vec::with_capacity(hosts.len() + 1);
116    out.push(KnownRemoteHost {
117        name: "localhost".into(),
118        in_hosts_file: true,
119        in_ssh_config: false,
120    });
121    out.extend(hosts);
122    out
123}

125/// Hostnames from `/etc/hosts` text (non-loopback, non-meta).
126#[must_use]
127pub fn parse_hosts_file_names(text: &str) -> Vec<String> {
128    let mut names = Vec::new();
129    for line in text.lines() {
130        let line = line.trim();
131        if line.is_empty() || line.starts_with('#') {
132            continue;
133        }
134        let mut parts = line.split_whitespace();
135        let Some(addr) = parts.next() else {
136            continue;
137        };
138        if is_skipped_hosts_address(addr) {
139            continue;
140        }
141        for name in parts {
142            let name = name.trim();
143            if name.is_empty() || is_skipped_hosts_name(name) {
144                continue;
145            }
146            if !names.iter().any(|n| n == name) {
147                names.push(name.to_owned());
148            }
149        }
150    }
151    names
152}

154/// Concrete `Host` aliases from OpenSSH config text (no wildcards).
155#[must_use]
156pub fn parse_ssh_config_host_aliases(text: &str) -> Vec<String> {
157    parse_ssh_config_hosts(text)
158        .into_iter()
159        .map(|h| h.alias)
160        .collect()
161}

163/// Parse `Host` / `HostName` blocks from OpenSSH config text.
164#[must_use]
165pub fn parse_ssh_config_hosts(text: &str) -> Vec<SshConfigHost> {
166    let mut out = Vec::new();
167    let mut current: Vec<String> = Vec::new();
168    let mut host_name: Option<String> = None;
169
170    let flush = |current: &mut Vec<String>, host_name: &mut Option<String>, out: &mut Vec<_>| {
171        if current.is_empty() {
172            *host_name = None;
173            return;
174        }
175        let hn = host_name.take();
176        for alias in current.drain(..) {
177            let resolved = hn.clone().unwrap_or_else(|| alias.clone());
178            out.push(SshConfigHost {
179                alias,
180                host_name: resolved,
181            });
182        }
183    };
184
185    for line in text.lines() {
186        let trimmed = line.trim();
187        if trimmed.is_empty() || trimmed.starts_with('#') {
188            continue;
189        }
190        let mut parts = trimmed.split_whitespace();
191        let Some(keyword) = parts.next() else {
192            continue;
193        };
194        if keyword.eq_ignore_ascii_case("host") {
195            flush(&mut current, &mut host_name, &mut out);
196            for token in parts {
197                if token.contains('*') || token.contains('?') || token.starts_with('!') {
198                    continue;
199                }
200                if !current.iter().any(|n| n == token) {
201                    current.push(token.to_owned());
202                }
203            }
204            continue;
205        }
206        if keyword.eq_ignore_ascii_case("hostname") {
207            if let Some(value) = parts.next() {
208                host_name = Some(value.to_owned());
209            }
210        }
211    }
212    flush(&mut current, &mut host_name, &mut out);
213    out
214}

216/// True when an SSH config entry targets the LAN (private IP or `/etc/hosts` name).
217#[must_use]
218pub fn ssh_host_is_lan(entry: &SshConfigHost, hosts_file_names: &[String]) -> bool {
219    if hosts_file_names
220        .iter()
221        .any(|n| n == &entry.alias || n == &entry.host_name)
222    {
223        return true;
224    }
225    is_private_or_link_local_ip(&entry.host_name)
226}

228fn list_known_remote_hosts_from(
229    hosts_path: &Path,
230    ssh_paths: &[PathBuf],
231) -> Result<Vec<KnownRemoteHost>> {
232    let hosts_text = read_optional_text(hosts_path)?;
233    let mut ssh_texts = Vec::new();
234    let mut ssh_refs: Vec<&str> = Vec::new();
235    for path in ssh_paths {
236        let text = read_optional_text(path)?;
237        if !text.is_empty() {
238            ssh_texts.push(text);
239        }
240    }
241    for text in &ssh_texts {
242        ssh_refs.push(text.as_str());
243    }
244    Ok(merge_known_remote_hosts(&hosts_text, &ssh_refs))
245}
246
247fn read_optional_text(path: &Path) -> Result<String> {
248    if !path.is_file() {
249        return Ok(String::new());
250    }
251    fs::read_to_string(path)
252        .map_err(|e| HortoError::msg(format!("failed to read {}: {e}", path.display())))
253}
254
255fn ssh_config_paths() -> Vec<PathBuf> {
256    let mut paths = Vec::new();
257    if let Ok(home) = std::env::var("HOME") {
258        let user = PathBuf::from(home).join(".ssh").join("config");
259        if user.is_file() {
260            paths.push(user);
261        }
262    }
263    let system = PathBuf::from("/etc/ssh/ssh_config");
264    if system.is_file() {
265        paths.push(system);
266    }
267    paths
268}

270fn is_skipped_hosts_address(addr: &str) -> bool {
271    let lower = addr.to_ascii_lowercase();
272    lower == "::1"
273        || lower.starts_with("127.")
274        || lower.starts_with("fe80:")
275        || lower.starts_with("ff0")
276}
277
278fn is_skipped_hosts_name(name: &str) -> bool {
279    matches!(
280        name.to_ascii_lowercase().as_str(),
281        "localhost" | "ip6-localhost" | "ip6-loopback" | "ip6-allnodes" | "ip6-allrouters"
282    )
283}
284
285fn is_private_or_link_local_ip(host: &str) -> bool {
286    let Ok(ip) = host.parse::<IpAddr>() else {
287        return false;
288    };
289    match ip {
290        IpAddr::V4(v4) => v4.is_private() || v4.is_link_local(),
291        IpAddr::V6(v6) => {
292            let octets = v6.octets();
293            // Unique local fc00::/7 or link-local fe80::/10
294            (octets[0] & 0xfe) == 0xfc || (octets[0] == 0xfe && (octets[1] & 0xc0) == 0x80)
295        }
296    }
297}

351    #[test]
352    fn merge_keeps_lan_drops_internet_ssh_alias() {
353        let hosts = "192.168.0.242 horto\n192.168.0.3 debby.local\n";
354        let ssh = "\
355Host horto
356  HostName horto
357  User greg
358
359Host verq_pre_prod
360  HostName verq.example.com
361  User groussac
362
363Host lab
364  HostName 10.0.0.5
365";
366        let list = merge_known_remote_hosts(hosts, &[ssh]);
367        let names: Vec<_> = list.iter().map(|h| h.name.as_str()).collect();
368        assert_eq!(names, vec!["localhost", "debby.local", "horto", "lab"]);
369        assert!(!names.contains(&"verq_pre_prod"));
370        let horto = list.iter().find(|h| h.name == "horto").unwrap();
371        assert!(horto.in_hosts_file && horto.in_ssh_config);
372        let lab = list.iter().find(|h| h.name == "lab").unwrap();
373        assert!(!lab.in_hosts_file && lab.in_ssh_config);
374        assert_eq!(list[0].name, "localhost");
375    }
