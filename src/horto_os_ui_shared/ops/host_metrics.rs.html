44/// Collect host metrics from the local machine.
45///
46/// Reads `/proc`, os-release, optional Armbian release, root `statvfs`, and
47/// `apt list --upgradable` (2 s timeout). Never runs `apt update`.
48#[must_use]
49pub fn collect_host_metrics() -> HostMetrics {
50    let (load_one, load_five, load_fifteen) =
51        parse_loadavg(&read_to_string("/proc/loadavg").unwrap_or_default());
52    let (disk_total_bytes, disk_used_bytes, disk_avail_bytes) = disk_root();
53    let os = parse_os_release(&read_to_string("/etc/os-release").unwrap_or_default());
54    let armbian =
55        parse_armbian_release(&read_to_string("/etc/armbian-release").unwrap_or_default());
56    HostMetrics {
57        cpu_percent: cpu_percent_sample(),
58        load_1: load_one,
59        load_5: load_five,
60        load_15: load_fifteen,
61        disk_total_bytes,
62        disk_used_bytes,
63        disk_avail_bytes,
64        os_pretty_name: os.pretty_name,
65        os_id: os.id,
66        os_version_id: os.version_id,
67        armbian_version: armbian.version,
68        armbian_board: armbian.board,
69        kernel: kernel_release(),
70        apt_upgradable: apt_upgradable_count(),
71    }
72}

78/// Parse the aggregate `cpu` line counters from `/proc/stat` text.
79#[must_use]
80pub fn parse_proc_stat_cpu(stat: &str) -> Option<(u64, u64)> {
81    let line = stat.lines().find(|l| l.starts_with("cpu "))?;
82    let mut parts = line.split_whitespace().skip(1);
83    let mut values = [0_u64; 10];
84    for slot in &mut values {
85        *slot = parts.next()?.parse().ok()?;
86    }
87    // user nice system idle iowait irq softirq steal guest guest_nice
88    let idle = values[3].saturating_add(values[4]);
89    let total: u64 = values.iter().take(8).sum();
90    Some((idle, total))
91}

93/// CPU busy percent from two idle/total samples.
94#[must_use]
95pub fn cpu_percent_from_samples(first: (u64, u64), second: (u64, u64)) -> Option<f32> {
96    let idle_delta = second.0.saturating_sub(first.0);
97    let total_delta = second.1.saturating_sub(first.1);
98    if total_delta == 0 {
99        return None;
100    }
101    let busy_pct = total_delta.saturating_sub(idle_delta).saturating_mul(100) / total_delta;
102    let capped = busy_pct.min(100);
103    let as_u8 = u8::try_from(capped).unwrap_or(100);
104    Some(f32::from(as_u8))
105}

107fn cpu_percent_sample() -> Option<f32> {
108    let first = parse_proc_stat_cpu(&read_to_string("/proc/stat")?)?;
109    thread::sleep(Duration::from_millis(100));
110    let second = parse_proc_stat_cpu(&read_to_string("/proc/stat")?)?;
111    cpu_percent_from_samples(first, second)
112}

114/// Parse `/proc/loadavg` into 1 / 5 / 15 minute averages.
115#[must_use]
116pub fn parse_loadavg(raw: &str) -> (Option<f32>, Option<f32>, Option<f32>) {
117    let mut parts = raw.split_whitespace();
118    let a = parts.next().and_then(|s| s.parse().ok());
119    let b = parts.next().and_then(|s| s.parse().ok());
120    let c = parts.next().and_then(|s| s.parse().ok());
121    (a, b, c)
122}

124/// Parse key=value lines from `/etc/os-release` style text.
125#[must_use]
126pub fn parse_os_release(raw: &str) -> OsReleaseFields {
127    let map = parse_kv(raw);
128    OsReleaseFields {
129        pretty_name: map_get(&map, "PRETTY_NAME"),
130        id: map_get(&map, "ID"),
131        version_id: map_get(&map, "VERSION_ID"),
132    }

146/// Parse Armbian release file fields.
147#[must_use]
148pub fn parse_armbian_release(raw: &str) -> ArmbianReleaseFields {
149    let map = parse_kv(raw);
150    let board = map_get(&map, "BOARD_NAME").or_else(|| map_get(&map, "BOARD"));
151    ArmbianReleaseFields {
152        version: map_get(&map, "VERSION"),
153        board,
154    }
155}

166fn parse_kv(raw: &str) -> std::collections::HashMap<String, String> {
167    let mut map = std::collections::HashMap::new();
168    for line in raw.lines() {
169        let line = line.trim();
170        if line.is_empty() || line.starts_with('#') {
171            continue;
172        }
173        let Some((k, v)) = line.split_once('=') else {
174            continue;
175        };
176        let key = k.trim();
177        let mut val = v.trim();
178        if val.len() >= 2 && val.starts_with('"') && val.ends_with('"') {
179            val = &val[1..val.len() - 1];
180        }
181        if !key.is_empty() {
182            map.insert(key.to_owned(), val.to_owned());
183        }
184    }
185    map
186}
187
188fn map_get(map: &std::collections::HashMap<String, String>, key: &str) -> Option<String> {
189    map.get(key)
190        .map(|s| s.trim().to_owned())
191        .filter(|s| !s.is_empty())
192}

194fn disk_root() -> (Option<u64>, Option<u64>, Option<u64>) {
195    disk_statvfs(c"/")
196}

198fn disk_statvfs(path: &std::ffi::CStr) -> (Option<u64>, Option<u64>, Option<u64>) {
199    // SAFETY: path is a valid C string; `statvfs` writes into `buf` on success.
200    let mut buf = std::mem::MaybeUninit::<libc::statvfs>::uninit();
201    let rc = unsafe { libc::statvfs(path.as_ptr(), buf.as_mut_ptr()) };
202    if rc != 0 {
203        return (None, None, None);
204    }
205    let st = unsafe { buf.assume_init() };
206    let bsize = st.f_frsize;
207    let total = bsize.saturating_mul(st.f_blocks);
208    let avail = bsize.saturating_mul(st.f_bavail);
209    let used = total.saturating_sub(avail);
210    (Some(total), Some(used), Some(avail))
211}

213fn kernel_release() -> Option<String> {
214    let o = Command::new("uname").arg("-r").output().ok()?;
215    if !o.status.success() {
216        return None;
217    }
218    let s = String::from_utf8_lossy(&o.stdout).trim().to_owned();
219    if s.is_empty() {
220        None
221    } else {
222        Some(s)
223    }
224}

226fn apt_upgradable_count() -> Option<u32> {
227    let (tx, rx) = mpsc::channel();
228    thread::spawn(move || {
229        let out = Command::new("apt")
230            .args(["list", "--upgradable"])
231            .stdout(Stdio::piped())
232            .stderr(Stdio::null())
233            .output();
234        let _ = tx.send(out);
235    });
236    match rx.recv_timeout(Duration::from_secs(2)) {
237        Ok(Ok(o)) if o.status.success() => {
238            Some(count_apt_upgradable(&String::from_utf8_lossy(&o.stdout)))
239        }
240        _ => None,
241    }
242}

244/// Count package lines in `apt list --upgradable` stdout (skips `Listing...`).
245#[must_use]
246pub fn count_apt_upgradable(stdout: &str) -> u32 {
247    let n = stdout
248        .lines()
249        .map(str::trim)
250        .filter(|line| !line.is_empty())
251        .filter(|line| !line.starts_with("Listing"))
252        .filter(|line| line.contains('/'))
253        .count();
254    u32::try_from(n).unwrap_or(u32::MAX)
255}

261    #[test]
262    fn proc_stat_cpu_parses_and_delta() {
263        let a = "cpu  100 0 50 850 0 0 0 0 0 0\ncpu0 50 0 25 425 0 0 0 0 0 0\n";
264        let b = "cpu  120 0 60 870 0 0 0 0 0 0\n";
265        let first = parse_proc_stat_cpu(a).expect("first");
266        let second = parse_proc_stat_cpu(b).expect("second");
267        assert_eq!(first, (850, 1000));
268        assert_eq!(second, (870, 1050));
269        let pct = cpu_percent_from_samples(first, second).expect("pct");
270        // idle +20, total +50 → busy 60%
271        assert!((pct - 60.0).abs() < 0.01);
272    }
