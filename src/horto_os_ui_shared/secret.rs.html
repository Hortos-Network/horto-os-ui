1//! Secret handling helpers (sudo passwords): wipe memory, never log.
2
3/// Overwrite a secret [`String`]'s bytes then clear it.
4///
5/// Call after the last use of a password buffer (stdin feed, IPC copy).
6pub fn wipe_secret(s: &mut String) {
7    let mut bytes = std::mem::take(s).into_bytes();
8    bytes.fill(0);
9    drop(bytes);
10}
11
12/// Replace every occurrence of `secret` in `text` (when secret is non-empty).
13///
14/// Skips secrets shorter than 2 chars to avoid nuking common letters.
15#[must_use]
16pub fn redact_secret(text: &str, secret: &str) -> String {
17    if secret.len() < 2 {
18        return text.to_owned();
19    }
20    text.replace(secret, "********")
21}
