13/// Initialize Docker stack trees from embedded assets (`d1`).
14pub struct D1Docker;
15
16impl Step for D1Docker {
17    fn id(&self) -> &'static str {
18        "d1"
19    }
20    fn title(&self) -> &'static str {
21        "Initialize Docker stacks from embedded source"
22    }
23    fn reference_script(&self) -> &'static str {
24        "d1_docker_init.sh"
25    }
26    fn step_version(&self) -> u32 {
27        4
28    }
29    fn depends_on(&self) -> &'static [&'static str] {
30        &["d0"]
31    }
32    fn is_done(&self, ctx: &HostContext) -> bool {
33        ctx.paths.docker.join("dockge/compose.yaml").exists()
34            || ctx.paths.docker.join("stacks").is_dir()
35            || ctx.paths.docker.join("homepage/compose.yaml").exists()
36    }
37    fn plan(&self, ctx: &mut HostContext) -> Result<Vec<PlannedAction>> {
38        let npu = resolve_npu_type(ctx)?;
39        ctx.plan_action(format!(
40            "ensure {}",
41            ctx.paths.os_configuration_file().display()
42        ));
43        ctx.plan_action(format!(
44            "extract embedded docker_source/ (non-stacks) -> {}",
45            ctx.paths.docker.display()
46        ));
47        ctx.plan_action("merge stacks/common into docker tree");
48        match npu.as_str() {
49            "none" => ctx.plan_action("NPU_TYPE=none; skip NPU/GPU stack merge"),
50            other => ctx.plan_action(format!("merge stacks/{other} into docker tree")),
51        }
52        ctx.plan_action("copy homepage_assets/* into docker/assets");
53        ctx.plan_action("stage homepage/os-configuration.env beside compose");
54        ctx.plan_action("render {{VAR}} placeholders under docker tree");
55        if ctx.skip_piper_download() {
56            ctx.plan_action("skip piper model download (--skip-piper or Piper stack not selected)");
57        } else {
58            ctx.plan_action("download piper models (warn-only on failure)");
59        }
60        Ok(ctx.planned.clone())
61    }
62    fn apply(&self, ctx: &mut HostContext) -> Result<()> {
63        ensure_os_configuration(ctx)?;
64        let vars = load_render_vars(ctx)?;
65        let npu = resolve_npu_type(ctx)?;
66        let target = ctx.paths.docker.clone();
67        fs::ensure_dir(ctx, &target)?;
68
69        extract_non_stacks(ctx, &target)?;
70        merge_stack_prefix(ctx, &target, "stacks/common/")?;
71        if npu == "none" {
72            ctx.log("NPU_TYPE=none; skipped NPU/GPU stack merge");
73        } else {
74            let prefix = format!("stacks/{npu}/");
75            merge_stack_prefix(ctx, &target, &prefix)?;
76        }
77
78        copy_homepage_assets(ctx)?;
79        stage_homepage_os_configuration(ctx)?;
80
81        if ctx.is_dry_run() {
82            ctx.plan_action("render placeholders in docker tree");
83        } else {
84            render_tree(ctx, &target, &vars)?;
85        }
86
87        if ctx.skip_piper_download() || ctx.is_dry_run() {
88            ctx.log("Skipping piper model download.");
89        } else {
90            download_piper_models(ctx);
91        }
92
93        ctx.log(format!(
94            "d1 complete: docker source ready under {} (stacks not started; see d2)",
95            target.display()
96        ));
97        Ok(())
98    }
99}

101fn ensure_os_configuration(ctx: &mut HostContext) -> Result<()> {
102    let dest = ctx.paths.os_configuration_file();
103    if dest.exists() {
104        return Ok(());
105    }
106    let template = embed::get("config/os-configuration.env")
107        .ok_or_else(|| HortoError::EmbedMissing("config/os-configuration.env".into()))?;
108    fs::ensure_dir(ctx, &ctx.paths.active_setup.clone())?;
109    fs::write_file(ctx, &dest, &template)?;
110    if ctx.is_dry_run() {
111        ctx.plan_action(format!("write {}", dest.display()));
112    } else {
113        ctx.log(format!("wrote {}", dest.display()));
114    }
115    Ok(())
116}

118/// Map WIP / alias values onto folders under `docker_source/stacks/`.
119fn normalize_npu_type(raw: &str) -> String {
120    match raw.trim() {
121        "" | "none" => "none".into(),
122        "rkRK3588" | "rk3588" | "RK3588" => "rk3588".into(),
123        "rkRK3576" | "rk3576" | "RK3576" => "rk3576".into(),
124        "no_wyoming" => "no_wyoming".into(),
125        other => other.to_string(),
126    }
127}

129fn resolve_npu_type(ctx: &HostContext) -> Result<String> {
130    let path = ctx.paths.os_configuration_file();
131    let map = if path.exists() {
132        envfile::load(&path)?
133    } else if ctx.is_dry_run() {
134        let template = embed::get_str("config/os-configuration.env").unwrap_or_default();
135        envfile::parse(&template)
136    } else {
137        return Err(HortoError::msg(format!(
138            "missing {}; run setup so os-configuration is staged",
139            path.display()
140        )));
141    };
142    let raw = map.get("NPU_TYPE").map_or("none", String::as_str);
143    let npu = normalize_npu_type(raw);
144    if npu == "none" {
145        return Ok(npu);
146    }
147    let probe = format!("docker_source/stacks/{npu}/");
148    let exists = Assets::iter().any(|p| p.starts_with(&probe));
149    if !exists {
150        return Err(HortoError::msg(format!(
151            "NPU_TYPE={npu} has no embedded stacks/{npu}/ (fix OS_TYPE/NPU_TYPE in os-configuration.env)"
152        )));
153    }
154    Ok(npu)
155}

157fn extract_non_stacks(ctx: &mut HostContext, target: &Path) -> Result<()> {
158    for path in Assets::iter() {
159        let Some(rel) = path.strip_prefix("docker_source/") else {
160            continue;
161        };
162        if rel.is_empty() || rel.starts_with("stacks/") {
163            continue;
164        }
165        let dest = target.join(rel);
166        let data = embed::get(path.as_ref())
167            .ok_or_else(|| HortoError::msg(format!("missing embed {path}")))?;
168        fs::write_file(ctx, &dest, &data)?;
169    }
170    ctx.log(format!(
171        "Copied docker source tree into {} (excluded stacks/)",
172        target.display()
173    ));
174    Ok(())
175}

177fn merge_stack_prefix(ctx: &mut HostContext, target: &Path, stack_prefix: &str) -> Result<()> {
178    let embed_prefix = format!("docker_source/{stack_prefix}");
179    let mut count = 0u32;
180    for path in Assets::iter() {
181        let Some(rel) = path.strip_prefix(&embed_prefix) else {
182            continue;
183        };
184        if rel.is_empty() {
185            continue;
186        }
187        let dest = target.join(rel);
188        let data = embed::get(path.as_ref())
189            .ok_or_else(|| HortoError::msg(format!("missing embed {path}")))?;
190        fs::write_file(ctx, &dest, &data)?;
191        count += 1;
192    }
193    ctx.log(format!(
194        "Merged {stack_prefix} ({count} files) into {}",
195        target.display()
196    ));
197    Ok(())
198}

200fn load_render_vars(ctx: &HostContext) -> Result<BTreeMap<String, String>> {
201    if ctx.paths.full_env_file().exists() {
202        return envfile::load(&ctx.paths.full_env_file());
203    }
204    if ctx.paths.minimal_env_file().exists() {
205        return envfile::load(&ctx.paths.minimal_env_file());
206    }
207    if ctx.paths.os_configuration_file().exists() {
208        return envfile::load(&ctx.paths.os_configuration_file());
209    }
210    if ctx.is_dry_run() {
211        let mut demo = BTreeMap::new();
212        demo.insert("MY_HOSTNAME".into(), "horto-dryrun".into());
213        demo.insert("WIFI_INTERFACE".into(), "wlan0".into());
214        demo.insert("WIFI_SSID".into(), "Horto-IoT-LAN".into());
215        demo.insert("WIFI_PASSPHRASE".into(), String::new());
216        demo.insert("MY_URL".into(), "example.net".into());
217        return Ok(demo);
218    }
219    Err(HortoError::msg(
220        "no active setup file found; run s2 or m1 first",
221    ))
222}

224/// Homepage compose `env_file: os-configuration.env` expects the file beside the compose.
225fn stage_homepage_os_configuration(ctx: &mut HostContext) -> Result<()> {
226    let homepage_dir = ctx.paths.docker.join("homepage");
227    if !homepage_dir.is_dir() && !ctx.is_dry_run() {
228        return Ok(());
229    }
230    let src = ctx.paths.os_configuration_file();
231    if !src.is_file() && !ctx.is_dry_run() {
232        return Ok(());
233    }
234    let dest = homepage_dir.join("os-configuration.env");
235    if ctx.is_dry_run() {
236        ctx.plan_action(format!("copy {} -> {}", src.display(), dest.display()));
237        return Ok(());
238    }
239    fs::ensure_dir(ctx, &homepage_dir)?;
240    fs::copy_file(ctx, &src, &dest)?;
241    ctx.log(format!(
242        "Staged homepage env: {} -> {}",
243        src.display(),
244        dest.display()
245    ));
246    Ok(())
247}

249fn copy_homepage_assets(ctx: &mut HostContext) -> Result<()> {
250    let assets_target = ctx.paths.docker_assets();
251    fs::ensure_dir(ctx, &assets_target)?;
252    let mut copied = 0u32;
253    for path in Assets::iter() {
254        let Some(name) = path.strip_prefix("homepage_assets/") else {
255            continue;
256        };
257        if name.is_empty() || !name.starts_with("homepage") {
258            continue;
259        }
260        let data = embed::get(path.as_ref())
261            .ok_or_else(|| HortoError::msg(format!("missing embed {path}")))?;
262        let dest = assets_target.join(name);
263        fs::write_file(ctx, &dest, &data)?;
264        ctx.log(format!("Copied asset: {}", dest.display()));
265        copied += 1;
266    }
267    if copied == 0 {
268        ctx.log("No homepage* assets found; skipping asset copy.");
269    }
270    Ok(())
271}

273fn render_tree(
274    ctx: &mut HostContext,
275    root: &PathBuf,
276    vars: &BTreeMap<String, String>,
277) -> Result<()> {
278    for entry in walkdir::WalkDir::new(root)
279        .into_iter()
280        .filter_map(std::result::Result::ok)
281    {
282        if !entry.file_type().is_file() {
283            continue;
284        }
285        let path = entry.path();
286        let Ok(text) = std::fs::read_to_string(path) else {
287            continue;
288        };
289        if !template::has_unreplaced_placeholders(&text) {
290            continue;
291        }
292        let rendered = template::render(&text, vars);
293        std::fs::write(path, rendered)?;
294        ctx.log(format!("Rendered placeholders in: {}", path.display()));
295    }
296    Ok(())
297}

299fn download_piper_models(ctx: &mut HostContext) {
300    let piper_dir = ctx.paths.docker.join("docker_repos/piper");
301    if let Err(e) = std::fs::create_dir_all(&piper_dir) {
302        ctx.log(format!("warning: cannot create piper dir: {e}"));
303        return;
304    }
305    let url = "https://github.com/Hanzo-Huang/rk3576-home-assistant-voice/releases/download/models-v1/piper-rk3576-models.tar.gz";
306    let archive = piper_dir.join("piper-rk3576-models.tar.gz");
307    ctx.log("Downloading piper models...");
308    let status = Command::new("wget")
309        .args(["-O", archive.to_str().unwrap_or("/dev/null"), url])
310        .current_dir(&piper_dir)
311        .status();
312    match status {
313        Ok(s) if s.success() => extract_piper_archive(ctx, &archive),
314        _ => {
315            ctx.log(
316                "warning: piper model download failed or wget missing; continuing without models",
317            );
318        }
319    }
320}

322/// Extract a piper models tarball into the docker tree root.
323///
324/// Tip layout merges stack folders into `/srv/docker` (no `stacks/` parent), so
325/// archive members like `piper/model/…` land beside the piper compose project.
326fn extract_piper_archive(ctx: &mut HostContext, archive: &Path) {
327    let extract_root = ctx.paths.docker.clone();
328    if let Err(e) = std::fs::create_dir_all(&extract_root) {
329        ctx.log(format!(
330            "warning: cannot create piper extract dir {}: {e}",
331            extract_root.display()
332        ));
333        return;
334    }
335    let Some(archive_s) = archive.to_str() else {
336        ctx.log("warning: piper archive path is not UTF-8; continuing without models");
337        return;
338    };
339    let Some(root_s) = extract_root.to_str() else {
340        ctx.log("warning: piper extract path is not UTF-8; continuing without models");
341        return;
342    };
343    match Command::new("tar")
344        .args(["-xzf", archive_s, "-C", root_s])
345        .status()
346    {
347        Ok(s) if s.success() => ctx.log(format!(
348            "Piper models extracted under {}.",
349            extract_root.display()
350        )),
351        Ok(s) => ctx.log(format!(
352            "warning: piper model extract failed (tar exit {}); continuing without models",
353            s.code().unwrap_or(-1)
354        )),
355        Err(e) => ctx.log(format!(
356            "warning: piper model extract failed ({e}); continuing without models"
357        )),
358    }

844        let previous = std::env::var_os("PATH");
845        // SAFETY: single-threaded critical section under PIPER_PATH_LOCK for this process.

857            None => unsafe { std::env::remove_var("PATH") },
858        }
859
860        assert!(ctx
861            .logs
862            .iter()
863            .any(|l| { l.contains("piper model extract failed") && !l.contains("tar exit") }));
