95impl ConfirmKind {
96    fn title(&self) -> &'static str {
97        match self {
98            Self::DestructiveStep(_) => "Confirm destructive step",
99            Self::Reboot => "Confirm reboot",
100            Self::SaveToken(_) => "Save API token",
101            Self::RebootAfterApply => "Reboot after apply",
102            Self::InstallStatusApi { .. } => "Install status-api?",
103            Self::InstallMcp { .. } => "Install MCP?",
104        }
105    }
106
107    fn body(&self) -> String {
108        match self {
109            Self::DestructiveStep(id) => {
110                format!("Step {id} is destructive. Continue?")
111            }
112            Self::Reboot => "Reboot the box now?".into(),
113            Self::SaveToken(_) => {
114                "Save status-api bearer to ~/.config/horto-os-ui/api_token?".into()
115            }
116            Self::RebootAfterApply => "Remote apply succeeded. Reboot the box now?".into(),
117            Self::InstallStatusApi { remote } => {
118                if *remote {
119                    "Install status-api (systemd) on the remote box?".into()
120                } else {
121                    "Install status-api (systemd) on this host?".into()
122                }
123            }
124            Self::InstallMcp { remote, .. } => {
125                if *remote {
126                    "Install MCP (systemd) on the remote box?".into()
127                } else {
128                    "Install MCP (systemd) on this host?".into()
129                }
130            }
131        }
132    }
133}

549    fn s0_line(&self) -> String {

560    fn rebuild_remote_steps(&mut self, setup: Option<&horto_os_ui_shared::SetupStatusReport>) {
561        let mut lines = vec![self.s0_line()];
562        if let Some(report) = setup {
563            lines.extend(report.steps.iter().map(|s| {
564                format!(
565                    "{} | {} | {}{}",
566                    s.id,
567                    s.status,
568                    s.title,
569                    if s.destructive { " *" } else { "" }
570                )
571            }));
572        } else {
573            lines.extend(pipeline(self.kind).iter().map(|s| {
574                format!(
575                    "{} | pending | {}{}",
576                    s.id(),
577                    s.title(),
578                    if s.destructive() { " *" } else { "" }
579                )
580            }));
581        }
582        self.status_lines = lines;
583        self.select_smart_step();
584    }
585
586    fn select_smart_step(&mut self) {
587        if self.remote.is_some() && !self.cli_current {
588            self.step_state.select(Some(0));
589            return;
590        }
591        let start = if self.remote.is_some() { 1 } else { 0 };
592        for (i, line) in self.status_lines.iter().enumerate().skip(start) {
593            let status = line.split(" | ").nth(1).unwrap_or("");
594            if status == "pending" {
595                self.step_state.select(Some(i));
596                return;
597            }
598        }
599        if self.status_lines.len() > start {
600            self.step_state.select(Some(start));
601        } else {
602            self.step_state.select(Some(0));
603        }
604    }

647    /// Spawn a background remote probe if none is already running.
648    fn start_remote_probe(&mut self) {
649        let Some(opts) = self.remote_opts() else {
650            return;
651        };
652        if self.probe_inflight {
653            self.message = "Refresh already running".into();
654            return;
655        }
656        let host = opts.host.clone();
657        let full = self.kind != SetupKind::Minimal;
658        self.probe_inflight = true;
659        self.box_cli = BoxCliView::Probing;
660        self.surfaces = None;
661        self.rebuild_remote_steps(None);
662        self.message = format!("Refreshing {host}");
663        let tx = self.probe_tx.clone();
664        thread::spawn(move || {
665            let _ = tx.send(run_remote_probe(opts, full));
666        });
667    }
668
669    /// Apply any completed background probe without blocking.
670    fn poll_probe(&mut self) {
671        match self.probe_rx.try_recv() {
672            Ok(outcome) => {
673                self.probe_inflight = false;
674                self.apply_remote_probe(outcome);
675            }
676            Err(mpsc::TryRecvError::Empty | mpsc::TryRecvError::Disconnected) => {}
677        }
678    }
679
680    fn apply_remote_probe(&mut self, outcome: RemoteProbeOutcome) {
681        let host = outcome.host;
682        let ok = match outcome.result {
683            Ok(ok) => ok,
684            Err(e) => {
685                self.box_cli = BoxCliView::Known(RemoteBoxCliStatus::Unreachable);
686                self.cli_current = false;
687                self.surfaces = None;
688                self.push_log(format!("ERROR probe: {e}"));
689                self.message = format!("Refresh failed: {e}");
690                self.rebuild_remote_steps(None);
691                self.refresh_panel_text();
692                return;

810        if self.reboot_inflight {

835                        Ok(()) => {
836                            let _ = tx.send(RebootEvent::BoxBack);
837                        }
838                        Err(e) => {
839                            let _ = tx.send(RebootEvent::Failed(e));
840                        }
841                    }
842                }
843                Err(e) => {
844                    let _ = tx.send(RebootEvent::Failed(e.to_string()));

851        let Some(rx) = &self.reboot_rx else {

855            Ok(RebootEvent::Issued) => {

876                }

1583/// Restores the terminal on drop (normal exit, `?`, or unwind).
1584struct TerminalGuard;
1585
1586impl TerminalGuard {
1587    fn enter() -> io::Result<(Self, Terminal<CrosstermBackend<io::Stdout>>)> {
1588        enable_raw_mode()?;
1589        execute!(
1590            stdout(),
1591            EnterAlternateScreen,
1592            Hide,
1593            CtClear(ClearType::All),
1594            CtClear(ClearType::Purge)
1595        )?;
1596        let mut terminal = Terminal::new(CrosstermBackend::new(stdout()))?;
1597        terminal.clear()?;
1598        Ok((Self, terminal))
1599    }
1600}

1602impl Drop for TerminalGuard {
1603    fn drop(&mut self) {
1604        restore_terminal();
1605    }

1608/// Reset SGR, show cursor, leave alt buffer, clear leftover TUI cells.
1609fn hard_reset_tty() {
1610    let mut out = stdout();
1611    let _ = write!(
1612        out,
1613        "\x1b[0m\x1b[?25h\x1b[?1049l\x1b[?47l\x1b[2J\x1b[3J\x1b[H"
1614    );
1615    let _ = out.execute(CtClear(ClearType::All));
1616    let _ = out.execute(CtClear(ClearType::Purge));
1617    let _ = out.execute(Show);
1618    let _ = out.flush();
1619}

1621fn restore_terminal() {
1622    let _ = disable_raw_mode();
1623    let mut out = stdout();
1624    let _ = out.execute(LeaveAlternateScreen);
1625    let _ = out.flush();
1626    hard_reset_tty();
1627}

1642fn run_app(terminal: &mut Terminal<CrosstermBackend<io::Stdout>>, app: &mut App) -> Result<()> {
1643    let mut boot_remote_probe = app.is_remote();
1644    loop {
1645        if STOP.load(Ordering::SeqCst) {
1646            return Ok(());
1647        }
1648        app.poll_probe();
1649        app.poll_reboot();
1650        app.poll_fetch();
1651        terminal.draw(|f| ui(f, app))?;
1652        if boot_remote_probe {
1653            app.start_remote_probe();
1654            boot_remote_probe = false;
1655        }
1656        if !event::poll(std::time::Duration::from_millis(200))? {
1657            continue;
1658        }
1659        let Event::Key(key) = event::read()? else {
1660            continue;
1661        };
1662        if key.kind != KeyEventKind::Press {
1663            continue;
1664        }
1665        if ctrl_c_quit(key) {
1666            return Ok(());
1667        }
1668        if handle_modal_key(app, key) {
1669            continue;
1670        }
1671        if is_quit(key) {
1672            return Ok(());
1673        }
1674        if app.help_open {
1675            match key.code {
1676                KeyCode::Esc | KeyCode::Char('?') => {
1677                    app.help_open = false;
1678                }
1679                _ => {}
1680            }
1681            continue;
1682        }
1683        match key.code {
1684            KeyCode::Esc => return Ok(()),
1685            KeyCode::Char('?') => {
1686                app.help_open = true;
1687                app.message = "Help".into();
1688            }
1689            KeyCode::Tab => {
1690                app.apply = !app.apply;
1691                app.message = if app.apply {
1692                    "APPLY".into()
1693                } else {
1694                    "PLAN".into()
1695                };
1696                if app.remote.is_none() {
1697                    app.refresh();
1698                }
1699            }
1700            KeyCode::Char(d) if d.is_ascii_digit() => {
1701                if let Some(screen) = Screen::from_digit(d, app.is_remote()) {
1702                    app.select_screen(screen);
1703                }
1704            }
1705            KeyCode::Char('c') if app.screen == Screen::Logs => app.clear_logs(),
1706            KeyCode::Char('r') => {
1707                app.refresh();
1708                if app.remote.is_none() {
1709                    app.message = "Refreshed".into();
1710                }
1711            }
1712            KeyCode::Char('f') | KeyCode::Char('F') => {
1713                app.fetch_current_tab();
1714            }
1715            KeyCode::Char('a') => app.run_all(),
1716            KeyCode::Char('b') => app.run_backup_etc(),
1717            KeyCode::Char('B') => app.show_disk_backup_status(),
1718            KeyCode::Char('p') | KeyCode::Char('P') => {
1719                let next = if app.kind == SetupKind::Full {
1720                    SetupKind::Minimal
1721                } else {
1722                    SetupKind::Full
1723                };
1724                app.set_pipeline_kind_local(next);
1725            }
1726            KeyCode::Char('e') | KeyCode::Char('E') if app.screen == Screen::Ssh => {
1727                app.open_host_editor();
1728            }
1729            KeyCode::Char('i') | KeyCode::Char('I') if app.screen == Screen::Ssh => {
1730                app.install_ssh_key_action();
1731            }
1732            KeyCode::Enter if app.screen == Screen::Setup => {
1733                app.run_selected();
1734            }
1735            KeyCode::Enter
1736                if matches!(
1737                    app.screen,
1738                    Screen::Ssh
1739                        | Screen::Cli
1740                        | Screen::Api
1741                        | Screen::Mcp
1742                        | Screen::Overview
1743                        | Screen::Reboot
1744                ) =>
1745            {
1746                app.run_surface_enter();
1747            }
1748            KeyCode::Up | KeyCode::Char('k') => {
1749                if let Some(i) = app.step_state.selected() {
1750                    if i > 0 {
1751                        app.step_state.select(Some(i - 1));
1752                    }
1753                }
1754            }
1755            KeyCode::Down | KeyCode::Char('j') => {
1756                let len = app.status_lines.len();
1757                if let Some(i) = app.step_state.selected() {
1758                    if i + 1 < len {
1759                        app.step_state.select(Some(i + 1));
1760                    }
1761                }
1762            }
1763            KeyCode::Left => {
1764                app.prev_screen();
1765            }
1766            KeyCode::Right => {
1767                app.next_screen();
1768            }
1769            _ => {}
1770        }
1771    }
1772}

1774/// Handle confirm / text / secret modals. Returns true when the key was consumed.
1775fn handle_modal_key(app: &mut App, key: KeyEvent) -> bool {
1776    match app.modal {
1777        Some(Modal::TextHost(_)) => {
1778            let Some(Modal::TextHost(mut input)) = app.modal.take() else {
1779                return true;
1780            };
1781            match input.handle_key(key) {
1782                TextInputResult::Continue => {
1783                    app.modal = Some(Modal::TextHost(input));
1784                }
1785                TextInputResult::Submit(value) => {
1786                    app.apply_host_edit(value);
1787                }
1788                TextInputResult::Cancel => {
1789                    app.push_log("Host edit cancelled");
1790                }
1791            }
1792            true
1793        }
1794        Some(Modal::SudoPassword(_)) => {
1795            let Some(Modal::SudoPassword(mut input)) = app.modal.take() else {
1796                return true;
1797            };
1798            match input.handle_key(key) {
1799                TextInputResult::Continue => {
1800                    app.modal = Some(Modal::SudoPassword(input));
1801                }
1802                TextInputResult::Submit(password) => {
1803                    if password.is_empty() {
1804                        app.push_log("Reboot cancelled (empty password)");
1805                        app.note("Reboot cancelled");
1806                    } else {
1807                        app.do_reboot(password);
1808                    }
1809                }
1810                TextInputResult::Cancel => {
1811                    app.push_log("Reboot cancelled");
1812                    app.note("Reboot cancelled");
1813                }
1814            }
1815            true
1816        }
1817        Some(Modal::Rebooting) => {
1818            if key.code == KeyCode::Esc {
1819                app.reboot_cancel.store(true, Ordering::SeqCst);
1820                app.reboot_inflight = false;
1821                app.reboot_rx = None;
1822                app.modal = None;
1823            }
1824            true
1825        }
1826        Some(Modal::Confirm(_)) => {
1827            match confirm_key(key) {
1828                ConfirmResult::Yes => app.resolve_confirm_yes(),
1829                ConfirmResult::No => app.resolve_confirm_no(),
1830                ConfirmResult::Ignore => {}
1831            }
1832            true
1833        }
1834        None => false,
1835    }
1836}

1952fn pad_footer_line(line: Line<'static>, width: u16) -> Line<'static> {
1953    let width = width as usize;
1954    if width == 0 {
1955        return Line::default();
1956    }
1957    let used = line.width();
1958    if used > width {
1959        let s = line.to_string();
1960        let mut out = String::new();
1961        for ch in s.chars() {
1962            if out.chars().count() + 1 >= width {
1963                break;
1964            }
1965            out.push(ch);
1966        }
1967        if s.chars().count() > width {
1968            if out.chars().count() == width {
1969                out.pop();
1970            }
1971            out.push('…');
1972        }
1973        return Line::from(out);
1974    }
1975    if used < width {
1976        let mut line = line;
1977        line.spans.push(Span::raw(" ".repeat(width - used)));
1978        return line;
1979    }
1980    line
1981}

1983fn short_reboot_err(err: &str) -> String {
1984    let lower = err.to_ascii_lowercase();
1985    if lower.contains("sorry, try again") || lower.contains("incorrect password") {
1986        return "Reboot failed: wrong sudo password".into();
1987    }
1988    if lower.contains("no password was provided") || lower.contains("a terminal is required") {
1989        return "Reboot failed: sudo password not accepted".into();
1990    }
1991    let one_line: String = err
1992        .chars()
1993        .map(|c| if c == '\n' || c == '\r' { ' ' } else { c })
1994        .collect();
1995    let trimmed = one_line.trim();
1996    if trimmed.len() > 80 {
1997        format!("Reboot failed: {}…", &trimmed[..77])
1998    } else {
1999        format!("Reboot failed: {trimmed}")
2000    }

2003/// After reboot is accepted: wait until the box stops answering once, then until it replies again.
2004/// Esc sets `cancel`. Returns `Err("cancelled")` or timeout.
2005fn wait_until_box_replies(host: &str, cancel: &AtomicBool) -> std::result::Result<(), String> {
2006    let deadline = Instant::now() + Duration::from_secs(300);
2007    let mut saw_down = false;
2008    let mut interval = Duration::from_millis(500);
2009    let max_interval = Duration::from_secs(3);
2010
2011    while Instant::now() < deadline {
2012        if cancel.load(Ordering::SeqCst) {
2013            return Err("cancelled".into());
2014        }
2015        let up = ssh_host_up(host);
2016        if !saw_down {
2017            if !up {
2018                saw_down = true;
2019            }
2020        } else if up {
2021            return Ok(());
2022        }
2023        sleep_cancellable(interval, cancel)?;
2024        interval = (interval + Duration::from_millis(250)).min(max_interval);
2025    }
2026    Err("timed out waiting for box to reply".into())
2027}

2029fn ssh_host_up(host: &str) -> bool {
2030    std::process::Command::new("ssh")
2031        .args([
2032            "-o",
2033            "BatchMode=yes",
2034            "-o",
2035            "ConnectTimeout=3",
2036            "-o",
2037            "StrictHostKeyChecking=accept-new",
2038            host,
2039            "true",
2040        ])
2041        .stdout(std::process::Stdio::null())
2042        .stderr(std::process::Stdio::null())
2043        .status()
2044        .map(|s| s.success())
2045        .unwrap_or(false)
2046}

2048fn sleep_cancellable(total: Duration, cancel: &AtomicBool) -> std::result::Result<(), String> {
2049    let mut slept = Duration::ZERO;
2050    let slice = Duration::from_millis(100);
2051    while slept < total {
2052        if cancel.load(Ordering::SeqCst) {
2053            return Err("cancelled".into());
2054        }
2055        thread::sleep(slice.min(total - slept));
2056        slept += slice;
2057    }
2058    Ok(())
2059}
